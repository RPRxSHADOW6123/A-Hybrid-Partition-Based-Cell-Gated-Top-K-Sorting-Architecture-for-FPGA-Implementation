## TOP MODULE

module top_module (
    clk,
    rst,
    start_block,
    valid_in,
    data_in,
    end_block,
    valid_out,
    data_out,
    done
);

parameter DATA_WIDTH = 8;
parameter TOPK = 8;
parameter INDEX_WIDTH = 4;
parameter FLUSH_CYCLES = 12;

input rst;
input start_block;
input valid_in;
input [DATA_WIDTH-1:0] data_in;
input end_block;

output valid_out;
output [DATA_WIDTH-1:0] data_out;
output done;

wire [DATA_WIDTH-1:0] pivot;
wire pivot_valid;

wire left_valid;
wire right_valid;

wire [DATA_WIDTH-1:0] left_data;
wire [DATA_WIDTH-1:0] right_data;

wire [INDEX_WIDTH-1:0] left_count;
wire [INDEX_WIDTH-1:0] right_count;

wire merge_start;

wire left_drain_start;
wire right_drain_start;

wire [INDEX_WIDTH-1:0] left_drain_count;
wire [INDEX_WIDTH-1:0] right_drain_count;

wire left_drain_valid;
wire right_drain_valid;

wire [DATA_WIDTH-1:0] left_drain_data;
wire [DATA_WIDTH-1:0] right_drain_data;

wire left_drain_done;
wire right_drain_done;

partition_unit #(
    .DATA_WIDTH(DATA_WIDTH)
) PU (
    .clk(clk),
    .rst(rst),
    .start_block(start_block),
    .valid_in(valid_in),
    .data_in(data_in),
    .pivot(pivot),
    .pivot_valid(pivot_valid),
    .left_valid(left_valid),
    .right_valid(right_valid),
    .left_data(left_data),
    .right_data(right_data)

cell_gated_sorter #(

    .drain_valid(left_drain_valid),
    .drain_data(left_drain_data),

cell_gated_sorter #(
    .DATA_WIDTH(DATA_WIDTH),

## PARTITION UNIT

module partition_unit (
    clk,
    rst,
    start_block,
    valid_in,
    data_in,
    pivot,
    pivot_valid,
    left_valid,
    right_valid,
    left_data,
    right_data
);

parameter DATA_WIDTH = 8;

input clk;
input rst;
input start_block;
input valid_in;
input [DATA_WIDTH-1:0] data_in;

output reg [DATA_WIDTH-1:0] pivot;
output reg pivot_valid;
output reg left_valid;
output reg right_valid;
output reg [DATA_WIDTH-1:0] left_data;
output reg [DATA_WIDTH-1:0] right_data;

always @(posedge clk) begin
    if (rst) begin
        pivot       <= {DATA_WIDTH{1'b0}};
        pivot_valid <= 1'b0;
        left_valid  <= 1'b0;
        right_valid <= 1'b0;
        left_data   <= {DATA_WIDTH{1'b0}};
        right_data  <= {DATA_WIDTH{1'b0}};
    end
    else begin
        left_valid  <= 1'b0;
        right_valid <= 1'b0;

        if (start_block == 1'b1) begin
            pivot       <= {DATA_WIDTH{1'b0}};
            pivot_valid <= 1'b0;
            left_data   <= {DATA_WIDTH{1'b0}};
            right_data  <= {DATA_WIDTH{1'b0}};
        end
        else if (valid_in == 1'b1) begin
            if (pivot_valid == 1'b0) begin
                pivot       <= data_in;
                pivot_valid <= 1'b1;
            end
            else begin
                if (data_in < pivot) begin
                    left_data  <= data_in;
                    left_valid <= 1'b1;
                end
                else begin
                    right_data  <= data_in;
                    right_valid <= 1'b1;
                end
            end
        end
    end
end

endmodule


## CELL_gated Soter ( both left and right )

module cell_gated_sorter (
    clk,
    rst,
    clear,

    valid_in,
    data_in,

    count,

    drain_start,
    drain_count,
    drain_valid,
    drain_data,
    drain_done
);

parameter DATA_WIDTH = 8;
parameter TOPK = 8;
parameter INDEX_WIDTH = 4;

input clk;
input rst;
input clear;

input valid_in;
input [DATA_WIDTH-1:0] data_in;

output reg [INDEX_WIDTH-1:0] count;

input drain_start;
input [INDEX_WIDTH-1:0] drain_count;

output reg drain_valid;
output reg [DATA_WIDTH-1:0] drain_data;
output reg drain_done;

localparam [INDEX_WIDTH-1:0] TOPK_CONST = TOPK;

// value[0] = largest
// value[TOPK-1] = smallest among stored Top-K
(* control_set_remap = "yes" *)
reg [DATA_WIDTH-1:0] value [0:TOPK-1];

(* control_set_remap = "yes" *)
reg [DATA_WIDTH-1:0] pass_data [0:TOPK-2];

reg [DATA_WIDTH-1:0] value_d [0:TOPK-1];
reg [DATA_WIDTH-1:0] pass_data_d [0:TOPK-2];

(* control_set_remap = "yes" *)
reg cell_valid [0:TOPK-1];

(* control_set_remap = "yes" *)
reg pass_valid [0:TOPK-2];

reg cell_valid_d [0:TOPK-1];
reg pass_valid_d [0:TOPK-2];

reg drain_active;
reg [INDEX_WIDTH-1:0] drain_remaining;

reg drain_active_d;
reg [INDEX_WIDTH-1:0] drain_remaining_d;

reg [INDEX_WIDTH-1:0] count_d;
reg drain_valid_d;
reg [DATA_WIDTH-1:0] drain_data_d;
reg drain_done_d;

integer i;

always @(*) begin
    count_d           = count;
    drain_active_d    = drain_active;
    drain_remaining_d = drain_remaining;
    drain_valid_d     = 1'b0;
    drain_data_d      = drain_data;
    drain_done_d      = 1'b0;

    for (i = 0; i < TOPK; i = i + 1) begin
        value_d[i]      = value[i];
        cell_valid_d[i] = cell_valid[i];
    end

    for (i = 0; i < TOPK-1; i = i + 1) begin
        pass_data_d[i]  = pass_data[i];
        pass_valid_d[i] = 1'b0;
    end

    // Start drain
    if (drain_start == 1'b1) begin
        drain_active_d    = (drain_count != 0);
        drain_remaining_d = drain_count;
        drain_valid_d     = 1'b0;
        drain_done_d      = 1'b0;

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid_d[i] = 1'b0;
        end
    end

    // Serial-drain mode
    else if (drain_active == 1'b1) begin
        drain_valid_d = 1'b1;
        drain_data_d  = value[0];

        for (i = 0; i < TOPK-1; i = i + 1) begin
            value_d[i]      = value[i+1];
            cell_valid_d[i] = cell_valid[i+1];
        end

        value_d[TOPK-1]      = value[TOPK-1];
        cell_valid_d[TOPK-1] = 1'b0;

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid_d[i] = 1'b0;
        end

        if (count != 0)
            count_d = count - 1'b1;

        if (drain_remaining == 1) begin
            drain_active_d    = 1'b0;
            drain_remaining_d = {INDEX_WIDTH{1'b0}};
            drain_done_d      = 1'b1;
        end
        else begin
            drain_remaining_d = drain_remaining - 1'b1;
            drain_done_d      = 1'b0;
        end
    end

    // Normal insertion mode
    else begin
        if ((valid_in == 1'b1) && (count < TOPK_CONST))
            count_d = count + 1'b1;

        // Cell 0
        if (valid_in == 1'b1) begin
            if (cell_valid[0] == 1'b0) begin
                value_d[0] = data_in;
            end
            else if (data_in > value[0]) begin
                value_d[0]     = data_in;
                pass_data_d[0] = value[0];
            end
            else begin
                pass_data_d[0] = data_in;
            end
        end

        cell_valid_d[0] = cell_valid[0] | valid_in;

        if ((valid_in == 1'b1) && (cell_valid[0] == 1'b1))
            pass_valid_d[0] = 1'b1;
        else
            pass_valid_d[0] = 1'b0;

        // Middle cells
        for (i = 1; i < TOPK-1; i = i + 1) begin
            if (pass_valid[i-1] == 1'b1) begin
                if (cell_valid[i] == 1'b0) begin
                    value_d[i] = pass_data[i-1];
                end
                else if (pass_data[i-1] > value[i]) begin
                    value_d[i]     = pass_data[i-1];
                    pass_data_d[i] = value[i];
                end
                else begin
                    pass_data_d[i] = pass_data[i-1];
                end
            end

            cell_valid_d[i] = cell_valid[i] | pass_valid[i-1];

            if ((pass_valid[i-1] == 1'b1) && (cell_valid[i] == 1'b1))
                pass_valid_d[i] = 1'b1;
            else
                pass_valid_d[i] = 1'b0;
        end

        // Last cell
        if (pass_valid[TOPK-2] == 1'b1) begin
            if (cell_valid[TOPK-1] == 1'b0) begin
                value_d[TOPK-1] = pass_data[TOPK-2];
            end
            else if (pass_data[TOPK-2] > value[TOPK-1]) begin
                value_d[TOPK-1] = pass_data[TOPK-2];
            end
        end

        cell_valid_d[TOPK-1] = cell_valid[TOPK-1] | pass_valid[TOPK-2];
    end
end

// Datapath registers: no reset, no clear
always @(posedge clk) begin
    for (i = 0; i < TOPK; i = i + 1) begin
        value[i] <= value_d[i];
    end

    for (i = 0; i < TOPK-1; i = i + 1) begin
        pass_data[i] <= pass_data_d[i];
    end

    drain_data <= drain_data_d;
end

// Control registers only
always @(posedge clk) begin
    if (rst || clear) begin
        count           <= {INDEX_WIDTH{1'b0}};
        drain_active    <= 1'b0;
        drain_remaining <= {INDEX_WIDTH{1'b0}};
        drain_valid     <= 1'b0;
        drain_done      <= 1'b0;

        for (i = 0; i < TOPK; i = i + 1) begin
            cell_valid[i] <= 1'b0;
        end

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid[i] <= 1'b0;
        end
    end
    else begin
        count           <= count_d;
        drain_active    <= drain_active_d;
        drain_remaining <= drain_remaining_d;
        drain_valid     <= drain_valid_d;
        drain_done      <= drain_done_d;

        for (i = 0; i < TOPK; i = i + 1) begin
            cell_valid[i] <= cell_valid_d[i];
        end

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid[i] <= pass_valid_d[i];
        end
    end
end

endmodule


## CONTROL UNIT

module control_unit (
    clk,
    rst,
    start_block,
    end_block,
    merge_start
);

parameter FLUSH_CYCLES = 12;

input clk;
input rst;
input start_block;
input end_block;

output reg merge_start;

reg flush_active;
reg [15:0] flush_count;

always @(posedge clk) begin
    if (rst) begin
        flush_active <= 1'b0;
        flush_count  <= 16'd0;
        merge_start  <= 1'b0;
    end
    else begin
        merge_start <= 1'b0;

        if (start_block == 1'b1) begin
            flush_active <= 1'b0;
            flush_count  <= 16'd0;
            merge_start  <= 1'b0;
        end
        else if (end_block == 1'b1) begin
            flush_active <= 1'b1;
            flush_count  <= 16'd0;
        end
        else if (flush_active == 1'b1) begin
            if (flush_count >= FLUSH_CYCLES) begin
                merge_start  <= 1'b1;
                flush_active <= 1'b0;
                flush_count  <= 16'd0;
            end
            else begin
                flush_count <= flush_count + 1'b1;
            end
        end
    end
end

endmodule


## MERGE UNIT

module merge_unit (
    clk,
    rst,
    start_merge,

    pivot,
    pivot_valid,

    left_count,
    right_count,

    left_drain_start,
    left_drain_count,
    left_drain_valid,
    left_drain_data,
    left_drain_done,

    right_drain_start,
    right_drain_count,
    right_drain_valid,
    right_drain_data,
    right_drain_done,

    valid_out,
    data_out,
    done
);

parameter DATA_WIDTH = 8;
parameter TOPK = 8;
parameter INDEX_WIDTH = 4;

localparam [INDEX_WIDTH-1:0] TOPK_CONST = TOPK;

input clk;
input rst;
input start_merge;

input [DATA_WIDTH-1:0] pivot;
input pivot_valid;

input [INDEX_WIDTH-1:0] left_count;
input [INDEX_WIDTH-1:0] right_count;

output reg left_drain_start;
output reg [INDEX_WIDTH-1:0] left_drain_count;
input left_drain_valid;
input [DATA_WIDTH-1:0] left_drain_data;
input left_drain_done;

output reg right_drain_start;
output reg [INDEX_WIDTH-1:0] right_drain_count;
input right_drain_valid;
input [DATA_WIDTH-1:0] right_drain_data;
input right_drain_done;

output reg valid_out;
output reg [DATA_WIDTH-1:0] data_out;
output reg done;

reg [2:0] state;

parameter IDLE          = 3'd0;
parameter START_RIGHT   = 3'd1;
parameter DRAIN_RIGHT   = 3'd2;
parameter COLLECT_PIVOT = 3'd3;
parameter START_LEFT    = 3'd4;
parameter DRAIN_LEFT    = 3'd5;
parameter OUTPUT_INIT   = 3'd6;
parameter OUTPUT_DATA   = 3'd7;

reg [DATA_WIDTH-1:0] outbuf [0:TOPK-1];

reg [INDEX_WIDTH-1:0] out_count;
reg [INDEX_WIDTH-1:0] out_index;

reg [INDEX_WIDTH-1:0] left_use;
reg [INDEX_WIDTH-1:0] right_use;
reg pivot_use;

reg [INDEX_WIDTH-1:0] left_use_calc;
reg [INDEX_WIDTH-1:0] right_use_calc;
reg pivot_use_calc;

reg [INDEX_WIDTH:0] remaining_calc;

integer i;

always @(*) begin
    right_use_calc = 0;
    left_use_calc  = 0;
    pivot_use_calc = 0;
    remaining_calc = 0;

    if (right_count >= TOPK_CONST) begin
        right_use_calc = TOPK_CONST;
        left_use_calc  = 0;
        pivot_use_calc = 0;
    end
    else begin
        right_use_calc = right_count;
        remaining_calc = TOPK - right_count;

        if ((pivot_valid == 1'b1) && (remaining_calc > 0)) begin
            pivot_use_calc = 1'b1;
            remaining_calc = remaining_calc - 1'b1;
        end

        if (left_count < remaining_calc)
            left_use_calc = left_count;
        else
            left_use_calc = remaining_calc[INDEX_WIDTH-1:0];
    end
end

always @(posedge clk) begin
    if (rst) begin
        state <= IDLE;

        left_drain_start  <= 1'b0;
        right_drain_start <= 1'b0;

        left_drain_count  <= 0;
        right_drain_count <= 0;

        left_use  <= 0;
        right_use <= 0;
        pivot_use <= 0;

        valid_out <= 1'b0;
        data_out  <= {DATA_WIDTH{1'b0}};
        done      <= 1'b0;

        out_count <= 0;
        out_index <= 0;

        for (i = 0; i < TOPK; i = i + 1) begin
            outbuf[i] <= {DATA_WIDTH{1'b0}};
        end
    end
    else begin
        valid_out <= 1'b0;
        done      <= 1'b0;

        left_drain_start  <= 1'b0;
        right_drain_start <= 1'b0;

        case (state)

            IDLE: begin
                if (start_merge == 1'b1) begin
                    left_use  <= left_use_calc;
                    right_use <= right_use_calc;
                    pivot_use <= pivot_use_calc;

                    out_count <= 0;
                    out_index <= 0;

                    for (i = 0; i < TOPK; i = i + 1) begin
                        outbuf[i] <= {DATA_WIDTH{1'b0}};
                    end

                    if (right_use_calc > 0) begin
                        state <= START_RIGHT;
                    end
                    else begin
                        state <= COLLECT_PIVOT;
                    end
                end
            end

            START_RIGHT: begin
                right_drain_count <= right_use;
                right_drain_start <= 1'b1;
                state <= DRAIN_RIGHT;
            end

            DRAIN_RIGHT: begin
                if (right_drain_valid == 1'b1) begin
                    if (out_count < TOPK_CONST) begin
                        outbuf[out_count] <= right_drain_data;
                        out_count <= out_count + 1'b1;
                    end
                end

                if (right_drain_done == 1'b1) begin
                    state <= COLLECT_PIVOT;
                end
            end

            COLLECT_PIVOT: begin
                if ((pivot_use == 1'b1) && (out_count < TOPK_CONST)) begin
                    outbuf[out_count] <= pivot;
                    out_count <= out_count + 1'b1;
                end

                if (left_use > 0) begin
                    state <= START_LEFT;
                end
                else begin
                    state <= OUTPUT_INIT;
                end
            end

            START_LEFT: begin
                left_drain_count <= left_use;
                left_drain_start <= 1'b1;
                state <= DRAIN_LEFT;
            end

            DRAIN_LEFT: begin
                if (left_drain_valid == 1'b1) begin
                    if (out_count < TOPK_CONST) begin
                        outbuf[out_count] <= left_drain_data;
                        out_count <= out_count + 1'b1;
                    end
                end

                if (left_drain_done == 1'b1) begin
                    state <= OUTPUT_INIT;
                end
            end

            OUTPUT_INIT: begin
                out_index <= out_count;
                state <= OUTPUT_DATA;
            end

            OUTPUT_DATA: begin
                if (out_index > 0) begin
                    valid_out <= 1'b1;
                    data_out  <= outbuf[out_index - 1'b1];
                    out_index <= out_index - 1'b1;
                end
                else begin
                    done  <= 1'b1;
                    state <= IDLE;
                end
            end

            default: begin
                state <= IDLE;
            end

        endcase
    end
end

endmodule


## TEST BENCH

module testbench;

reg clk;
reg rst;
reg start_block;
reg valid_in;
reg end_block;
reg [7:0] data_in;

wire valid_out;
wire [7:0] data_out;
wire done;

parameter TOTAL_INPUTS = 1024;
parameter PRINT_ALL_INPUTS = 0;

integer i;
integer j;
integer seed;
integer random_value;
integer rand8;

integer input_count;
integer output_count;

integer exp_count;
integer exp_top [0:7];
integer hw_out  [0:7];

integer temp;
integer error_count;

real start_time;
real end_time;
real latency;
real clock_cycles;

top_module DUT (
    .clk(clk),
    .rst(rst),
    .start_block(start_block),
    .valid_in(valid_in),
    .data_in(data_in),
    .end_block(end_block),
    .valid_out(valid_out),
    .data_out(data_out),
    .done(done)
);

// DO NOT CHANGE CLOCK
always #5 clk = ~clk;

task sort_expected;
    integer a;
    integer b;
    begin
        for (a = 0; a < exp_count; a = a + 1) begin
            for (b = 0; b < exp_count - 1; b = b + 1) begin
                if (exp_top[b] > exp_top[b+1]) begin
                    temp = exp_top[b];
                    exp_top[b] = exp_top[b+1];
                    exp_top[b+1] = temp;
                end
            end
        end
    end
endtask

task insert_expected;
    input integer value;
    begin
        if (exp_count < 8) begin
            exp_top[exp_count] = value;
            exp_count = exp_count + 1;
            sort_expected;
        end
        else begin
            if (value > exp_top[0]) begin
                exp_top[0] = value;
                sort_expected;
            end
        end
    end
endtask

initial begin
    clk = 0;
    rst = 1;
    start_block = 0;
    valid_in = 0;
    end_block = 0;
    data_in = 0;

    input_count = 0;
    output_count = 0;
    exp_count = 0;
    error_count = 0;

    seed = 12345;

    for (i = 0; i < 8; i = i + 1) begin
        exp_top[i] = 0;
        hw_out[i] = 0;
    end

    $display("========================================");
    $display("RECTIFIED STREAMING TOP-8 TEST");
    $display("TOTAL INPUTS = %0d", TOTAL_INPUTS);
    $display("CLOCK        = 10 ns");
    $display("OUTPUT ORDER = ASCENDING TOP-8");
    $display("========================================");

    #20;
    rst = 0;

    @(negedge clk);
    start_block = 1'b1;

    @(negedge clk);
    start_block = 1'b0;

    start_time = $realtime;

    // Corrected input timing:
    // data is assigned before the next positive edge captures it.
    valid_in = 1'b1;

    for (i = 0; i < TOTAL_INPUTS; i = i + 1) begin
        random_value = $random(seed);
        rand8 = random_value & 255;

        data_in = rand8;

        input_count = input_count + 1;
        insert_expected(rand8);

        if (PRINT_ALL_INPUTS == 1) begin
            $display("[%0.0f ns] STREAM INPUT %0d = %0d",
                     $realtime,
                     input_count,
                     rand8);
        end

        @(negedge clk);
    end

    valid_in = 1'b0;
    data_in = 0;

    @(negedge clk);
    end_block = 1'b1;

    $display("========================================");
    $display("[%0.0f ns] END BLOCK SENT", $realtime);
    $display("========================================");

    @(negedge clk);
    end_block = 1'b0;
end

always @(posedge clk) begin
    if (valid_out == 1'b1) begin
        if (output_count < 8) begin
            hw_out[output_count] = data_out;
        end

        output_count = output_count + 1;

        $display("[%0.0f ns] TOP8 OUTPUT %0d = %0d",
                 $realtime,
                 output_count,
                 data_out);
    end
end

always @(posedge clk) begin
    if (done == 1'b1) begin
        end_time = $realtime;
        latency = end_time - start_time;
        clock_cycles = latency / 10.0;

        $display("========================================");
        $display("EXPECTED TOP-8 ASCENDING");
        $display("========================================");

        for (j = 0; j < 8; j = j + 1) begin
            $display("EXPECTED %0d = %0d", j + 1, exp_top[j]);
        end

        $display("========================================");
        $display("HARDWARE TOP-8 ASCENDING");
        $display("========================================");

        for (j = 0; j < 8; j = j + 1) begin
            $display("HW OUTPUT %0d = %0d", j + 1, hw_out[j]);
        end

        error_count = 0;

        for (j = 0; j < 8; j = j + 1) begin
            if (hw_out[j] != exp_top[j]) begin
                error_count = error_count + 1;
            end
        end

        if (output_count != 8) begin
            error_count = error_count + 1;
            $display("OUTPUT COUNT ERROR: output_count = %0d, expected = 8", output_count);
        end

        $display("========================================");
        $display("PERFORMANCE REPORT");
        $display("========================================");
        $display("TOTAL INPUTS  = %0d", input_count);
        $display("TOTAL OUTPUTS = %0d", output_count);
        $display("START TIME    = %0.0f ns", start_time);
        $display("END TIME      = %0.0f ns", end_time);
        $display("LATENCY       = %0.0f ns", latency);
        $display("CLOCK PERIOD  = 10 ns");
        $display("CLOCK CYCLES  = %0.0f", clock_cycles);

        if (error_count == 0) begin
            $display("RESULT        = PASS");
        end
        else begin
            $display("RESULT        = FAIL");
            $display("ERROR COUNT   = %0d", error_count);
        end

        $display("========================================");
        $display("SIMULATION FINISHED");
        $display("========================================");

        $finish;
    end
end

initial begin
    #20000;
    $display("========================================");
    $display("ERROR: SIMULATION TIMEOUT");
    $display("done signal not received");
    $display("========================================");
    $finish;
end
