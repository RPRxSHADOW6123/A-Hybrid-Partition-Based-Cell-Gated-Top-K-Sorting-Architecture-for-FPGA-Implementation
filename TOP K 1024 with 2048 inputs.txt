## PARTITION UNIT 

module partition_unit (
    clk,
    rst,
    start_block,
    valid_in,
    data_in,
    pivot,
    pivot_valid,
    left_valid,
    right_valid,
    left_data,
    right_data
);

parameter DATA_WIDTH = 8;

input clk;
input rst;
input start_block;
input valid_in;
input [DATA_WIDTH-1:0] data_in;

output reg [DATA_WIDTH-1:0] pivot;
output reg pivot_valid;
output reg left_valid;
output reg right_valid;
output reg [DATA_WIDTH-1:0] left_data;
output reg [DATA_WIDTH-1:0] right_data;

always @(posedge clk) begin
    if (rst) begin
        pivot       <= {DATA_WIDTH{1'b0}};
        pivot_valid <= 1'b0;
        left_valid  <= 1'b0;
        right_valid <= 1'b0;
        left_data   <= {DATA_WIDTH{1'b0}};
        right_data  <= {DATA_WIDTH{1'b0}};
    end
    else begin
        left_valid  <= 1'b0;
        right_valid <= 1'b0;

        if (start_block) begin
            pivot       <= {DATA_WIDTH{1'b0}};
            pivot_valid <= 1'b0;
            left_data   <= {DATA_WIDTH{1'b0}};
            right_data  <= {DATA_WIDTH{1'b0}};
        end
        else if (valid_in) begin
            if (!pivot_valid) begin
                pivot       <= data_in;
                pivot_valid <= 1'b1;
            end
            else begin
                if (data_in < pivot) begin
                    left_data  <= data_in;
                    left_valid <= 1'b1;
                end
                else begin
                    right_data  <= data_in;
                    right_valid <= 1'b1;
                end
            end
        end
    end
end

endmodule



## CELL GATED SORTER ( BOTH LEFT AND RIGHT SORTER)

module cell_gated_sorter (
    clk,
    rst,
    clear,

    valid_in,
    data_in,

    count,

    drain_start,
    drain_count,
    drain_valid,
    drain_data,
    drain_done
);

parameter DATA_WIDTH  = 8;
parameter TOPK        = 1024;
parameter INDEX_WIDTH = 11;

input clk;
input rst;
input clear;

input valid_in;
input [DATA_WIDTH-1:0] data_in;

output reg [INDEX_WIDTH-1:0] count;

input drain_start;
input [INDEX_WIDTH-1:0] drain_count;

output reg drain_valid;
output reg [DATA_WIDTH-1:0] drain_data;
output reg drain_done;

localparam [INDEX_WIDTH-1:0] TOPK_CONST = TOPK;

// ============================================================
// Large datapath registers
// No reset/clear on these.
// This avoids thousands of reset-based control sets.
// ============================================================

(* control_set_remap = "yes" *)
reg [DATA_WIDTH-1:0] value [0:TOPK-1];

(* control_set_remap = "yes" *)
reg [DATA_WIDTH-1:0] pass_data [0:TOPK-2];

reg [DATA_WIDTH-1:0] value_d [0:TOPK-1];
reg [DATA_WIDTH-1:0] pass_data_d [0:TOPK-2];


// ============================================================
// Control/valid registers
// These are small compared to datapath and can be reset.
// ============================================================

(* control_set_remap = "yes" *)
reg cell_valid [0:TOPK-1];

(* control_set_remap = "yes" *)
reg pass_valid [0:TOPK-2];

reg cell_valid_d [0:TOPK-1];
reg pass_valid_d [0:TOPK-2];

reg drain_active;
reg [INDEX_WIDTH-1:0] drain_remaining;

reg drain_active_d;
reg [INDEX_WIDTH-1:0] drain_remaining_d;

reg [INDEX_WIDTH-1:0] count_d;
reg drain_valid_d;
reg [DATA_WIDTH-1:0] drain_data_d;
reg drain_done_d;

integer i;


// ============================================================
// Next-state combinational logic
// ============================================================

always @(*) begin
    // Default hold
    count_d           = count;
    drain_active_d    = drain_active;
    drain_remaining_d = drain_remaining;
    drain_valid_d     = 1'b0;
    drain_data_d      = drain_data;
    drain_done_d      = 1'b0;

    for (i = 0; i < TOPK; i = i + 1) begin
        value_d[i]      = value[i];
        cell_valid_d[i] = cell_valid[i];
    end

    for (i = 0; i < TOPK-1; i = i + 1) begin
        pass_data_d[i]  = pass_data[i];
        pass_valid_d[i] = 1'b0;
    end


    // ========================================================
    // Start drain
    // ========================================================

    if (drain_start == 1'b1) begin
        drain_active_d    = (drain_count != 0);
        drain_remaining_d = drain_count;
        drain_valid_d     = 1'b0;
        drain_done_d      = 1'b0;

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid_d[i] = 1'b0;
        end
    end


    // ========================================================
    // Drain mode
    // Output always comes from value[0].
    // No 1024-to-1 random read mux.
    // ========================================================

    else if (drain_active == 1'b1) begin
        drain_valid_d = 1'b1;
        drain_data_d  = value[0];

        for (i = 0; i < TOPK-1; i = i + 1) begin
            value_d[i]      = value[i+1];
            cell_valid_d[i] = cell_valid[i+1];
        end

        value_d[TOPK-1]      = value[TOPK-1];
        cell_valid_d[TOPK-1] = 1'b0;

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid_d[i] = 1'b0;
        end

        if (count != 0)
            count_d = count - 1'b1;
        else
            count_d = count;

        if (drain_remaining == 1) begin
            drain_active_d    = 1'b0;
            drain_remaining_d = {INDEX_WIDTH{1'b0}};
            drain_done_d      = 1'b1;
        end
        else begin
            drain_remaining_d = drain_remaining - 1'b1;
            drain_done_d      = 1'b0;
        end
    end


    // ========================================================
    // Normal insertion mode
    // Cell-gated top-K insertion pipeline
    // value[0] = largest
    // value[TOPK-1] = smallest among stored top-K
    // ========================================================

    else begin

        if ((valid_in == 1'b1) && (count < TOPK_CONST))
            count_d = count + 1'b1;
        else
            count_d = count;


        // ----------------------------
        // Cell 0
        // ----------------------------
        if (valid_in == 1'b1) begin
            if (cell_valid[0] == 1'b0) begin
                value_d[0] = data_in;
            end
            else if (data_in > value[0]) begin
                value_d[0]     = data_in;
                pass_data_d[0] = value[0];
            end
            else begin
                pass_data_d[0] = data_in;
            end
        end

        cell_valid_d[0] = cell_valid[0] | valid_in;

        if ((valid_in == 1'b1) && (cell_valid[0] == 1'b1))
            pass_valid_d[0] = 1'b1;
        else
            pass_valid_d[0] = 1'b0;


        // ----------------------------
        // Middle cells
        // ----------------------------
        for (i = 1; i < TOPK-1; i = i + 1) begin
            if (pass_valid[i-1] == 1'b1) begin
                if (cell_valid[i] == 1'b0) begin
                    value_d[i] = pass_data[i-1];
                end
                else if (pass_data[i-1] > value[i]) begin
                    value_d[i]     = pass_data[i-1];
                    pass_data_d[i] = value[i];
                end
                else begin
                    pass_data_d[i] = pass_data[i-1];
                end
            end

            cell_valid_d[i] = cell_valid[i] | pass_valid[i-1];

            if ((pass_valid[i-1] == 1'b1) && (cell_valid[i] == 1'b1))
                pass_valid_d[i] = 1'b1;
            else
                pass_valid_d[i] = 1'b0;
        end


        // ----------------------------
        // Last cell
        // ----------------------------
        if (pass_valid[TOPK-2] == 1'b1) begin
            if (cell_valid[TOPK-1] == 1'b0) begin
                value_d[TOPK-1] = pass_data[TOPK-2];
            end
            else if (pass_data[TOPK-2] > value[TOPK-1]) begin
                value_d[TOPK-1] = pass_data[TOPK-2];
            end
        end

        cell_valid_d[TOPK-1] = cell_valid[TOPK-1] | pass_valid[TOPK-2];
    end
end


// ============================================================
// Datapath registers
// No reset.
// No clear.
// Old data is ignored when valid bits are zero.
// ============================================================

always @(posedge clk) begin
    for (i = 0; i < TOPK; i = i + 1) begin
        value[i] <= value_d[i];
    end

    for (i = 0; i < TOPK-1; i = i + 1) begin
        pass_data[i] <= pass_data_d[i];
    end

    drain_data <= drain_data_d;
end


// ============================================================
// Control registers
// Reset/clear only control state and valid bits.
// ============================================================

always @(posedge clk) begin
    if (rst || clear) begin
        count           <= {INDEX_WIDTH{1'b0}};
        drain_active    <= 1'b0;
        drain_remaining <= {INDEX_WIDTH{1'b0}};
        drain_valid     <= 1'b0;
        drain_done      <= 1'b0;

        for (i = 0; i < TOPK; i = i + 1) begin
            cell_valid[i] <= 1'b0;
        end

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid[i] <= 1'b0;
        end
    end
    else begin
        count           <= count_d;
        drain_active    <= drain_active_d;
        drain_remaining <= drain_remaining_d;
        drain_valid     <= drain_valid_d;
        drain_done      <= drain_done_d;

        for (i = 0; i < TOPK; i = i + 1) begin
            cell_valid[i] <= cell_valid_d[i];
        end

        for (i = 0; i < TOPK-1; i = i + 1) begin
            pass_valid[i] <= pass_valid_d[i];
        end
    end
end

endmodule



## CONTROL UNIT

module control_unit (
    clk,
    rst,
    start_block,
    end_block,
    merge_start
);

parameter FLUSH_CYCLES = 1030;

input clk;
input rst;
input start_block;
input end_block;

output reg merge_start;

reg flush_active;
reg [15:0] flush_count;

always @(posedge clk) begin
    if (rst) begin
        flush_active <= 1'b0;
        flush_count  <= 16'd0;
        merge_start  <= 1'b0;
    end
    else begin
        merge_start <= 1'b0;

        if (start_block) begin
            flush_active <= 1'b0;
            flush_count  <= 16'd0;
            merge_start  <= 1'b0;
        end
        else if (end_block) begin
            flush_active <= 1'b1;
            flush_count  <= 16'd0;
        end
        else if (flush_active) begin
            if (flush_count >= FLUSH_CYCLES) begin
                merge_start  <= 1'b1;
                flush_active <= 1'b0;
                flush_count  <= 16'd0;
            end
            else begin
                flush_count <= flush_count + 1'b1;
            end
        end
    end
end

endmodule



## MERGE UNIT 

module merge_unit (
    clk,
    rst,
    start_merge,

    pivot,
    pivot_valid,

    left_count,
    right_count,

    left_drain_start,
    left_drain_count,
    left_drain_valid,
    left_drain_data,
    left_drain_done,

    right_drain_start,
    right_drain_count,
    right_drain_valid,
    right_drain_data,
    right_drain_done,

    valid_out,
    data_out,
    done
);

parameter DATA_WIDTH  = 8;
parameter TOPK        = 1024;
parameter INDEX_WIDTH = 11;

localparam [INDEX_WIDTH-1:0] TOPK_CONST = TOPK;

input clk;
input rst;
input start_merge;

input [DATA_WIDTH-1:0] pivot;
input pivot_valid;

input [INDEX_WIDTH-1:0] left_count;
input [INDEX_WIDTH-1:0] right_count;

output reg left_drain_start;
output reg [INDEX_WIDTH-1:0] left_drain_count;
input left_drain_valid;
input [DATA_WIDTH-1:0] left_drain_data;
input left_drain_done;

output reg right_drain_start;
output reg [INDEX_WIDTH-1:0] right_drain_count;
input right_drain_valid;
input [DATA_WIDTH-1:0] right_drain_data;
input right_drain_done;

output reg valid_out;
output reg [DATA_WIDTH-1:0] data_out;
output reg done;

reg [2:0] state;

parameter IDLE         = 3'd0;
parameter DRAIN_RIGHT  = 3'd1;
parameter OUTPUT_PIVOT = 3'd2;
parameter START_LEFT   = 3'd3;
parameter DRAIN_LEFT   = 3'd4;
parameter DONE_STATE   = 3'd5;

reg [INDEX_WIDTH-1:0] left_use;
reg [INDEX_WIDTH-1:0] right_use;
reg pivot_use;

reg [INDEX_WIDTH-1:0] left_use_calc;
reg [INDEX_WIDTH-1:0] right_use_calc;
reg pivot_use_calc;

reg [INDEX_WIDTH:0] remaining_calc;

always @(*) begin
    right_use_calc = 0;
    left_use_calc  = 0;
    pivot_use_calc = 0;
    remaining_calc = 0;

    if (right_count >= TOPK_CONST) begin
        right_use_calc = TOPK_CONST;
        left_use_calc  = 0;
        pivot_use_calc = 0;
    end
    else begin
        right_use_calc = right_count;
        remaining_calc = TOPK_CONST - right_count;

        if ((pivot_valid == 1'b1) && (remaining_calc > 0)) begin
            pivot_use_calc = 1'b1;
            remaining_calc = remaining_calc - 1'b1;
        end

        if (left_count < remaining_calc)
            left_use_calc = left_count;
        else
            left_use_calc = remaining_calc[INDEX_WIDTH-1:0];
    end
end

always @(posedge clk) begin
    if (rst) begin
        state <= IDLE;

        left_use  <= 0;
        right_use <= 0;
        pivot_use <= 0;

        left_drain_start  <= 1'b0;
        right_drain_start <= 1'b0;

        left_drain_count  <= 0;
        right_drain_count <= 0;

        valid_out <= 1'b0;
        data_out  <= {DATA_WIDTH{1'b0}};
        done      <= 1'b0;
    end
    else begin
        valid_out <= 1'b0;
        done      <= 1'b0;

        left_drain_start  <= 1'b0;
        right_drain_start <= 1'b0;

        case (state)

            IDLE: begin
                if (start_merge) begin
                    left_use  <= left_use_calc;
                    right_use <= right_use_calc;
                    pivot_use <= pivot_use_calc;

                    if (right_use_calc > 0) begin
                        right_drain_count <= right_use_calc;
                        right_drain_start <= 1'b1;
                        state <= DRAIN_RIGHT;
                    end
                    else if (pivot_use_calc) begin
                        state <= OUTPUT_PIVOT;
                    end
                    else if (left_use_calc > 0) begin
                        state <= START_LEFT;
                    end
                    else begin
                        state <= DONE_STATE;
                    end
                end
            end

            DRAIN_RIGHT: begin
                if (right_drain_valid) begin
                    valid_out <= 1'b1;
                    data_out  <= right_drain_data;
                end

                if (right_drain_done) begin
                    if (pivot_use) begin
                        state <= OUTPUT_PIVOT;
                    end
                    else if (left_use > 0) begin
                        state <= START_LEFT;
                    end
                    else begin
                        state <= DONE_STATE;
                    end
                end
            end

            OUTPUT_PIVOT: begin
                valid_out <= 1'b1;
                data_out  <= pivot;

                if (left_use > 0) begin
                    state <= START_LEFT;
                end
                else begin
                    state <= DONE_STATE;
                end
            end

            START_LEFT: begin
                left_drain_count <= left_use;
                left_drain_start <= 1'b1;
                state <= DRAIN_LEFT;
            end

            DRAIN_LEFT: begin
                if (left_drain_valid) begin
                    valid_out <= 1'b1;
                    data_out  <= left_drain_data;
                end

                if (left_drain_done) begin
                    state <= DONE_STATE;
                end
            end

            DONE_STATE: begin
                done  <= 1'b1;
                state <= IDLE;
            end

            default: begin
                state <= IDLE;
            end

        endcase
    end
end

endmodule

        input_count = input_count + 1;

        if (PRINT_INPUTS == 1) begin
            $display("[%0.0f ns] STREAM INPUT %0d = %0d",
                     $realtime,
                     input_count,
                     rand8);
        end
    end

    @(negedge clk);
    valid_in = 1'b0;
    data_in = 0;

    @(negedge clk);
    end_block = 1'b1;

    $display("========================================");
    $display("[%0.0f ns] END BLOCK SENT", $realtime);
    $display("========================================");

    @(negedge clk);
    end_block = 1'b0;
end

always @(posedge clk) begin
    if (valid_out == 1'b1) begin
        if (output_count < TOPK) begin
            hw_out[output_count] = data_out;
        end

        output_count = output_count + 1;

        if (PRINT_OUTPUTS == 1) begin
            $display("[%0.0f ns] TOP%0d OUTPUT %0d = %0d",
                     $realtime,
                     TOPK,
                     output_count,
                     data_out);
        end
    end

    if (done == 1'b1) begin
        end_time = $realtime;
        latency = end_time - start_time;
        clock_cycles = latency / CLK_PERIOD;

        build_expected_topk;

        error_count = 0;

        for (j = 0; j < TOPK; j = j + 1) begin
            if (hw_out[j] != expected[j]) begin
                error_count = error_count + 1;
            end
        end

        stream_throughput_mbps = (DATA_WIDTH * 1000.0) / CLK_PERIOD;
        stream_throughput_gbps = stream_throughput_mbps / 1000.0;

        overall_throughput_mbps = (TOTAL_INPUTS * DATA_WIDTH * 1000.0) / latency;
        overall_throughput_gbps = overall_throughput_mbps / 1000.0;

        $display("========================================");
        $display("FIRST 16 EXPECTED VALUES");
        $display("========================================");

        for (j = 0; j < 16; j = j + 1) begin
            $display("EXPECTED %0d = %0d", j + 1, expected[j]);
        end

        $display("========================================");
        $display("FIRST 16 HARDWARE VALUES");
        $display("========================================");

        for (j = 0; j < 16; j = j + 1) begin
            $display("HW OUTPUT %0d = %0d", j + 1, hw_out[j]);
        end

        $display("========================================");
        $display("LAST 16 EXPECTED VALUES");
        $display("========================================");

        for (j = TOPK-16; j < TOPK; j = j + 1) begin
            $display("EXPECTED %0d = %0d", j + 1, expected[j]);
        end

        $display("========================================");
        $display("LAST 16 HARDWARE VALUES");
        $display("========================================");

        for (j = TOPK-16; j < TOPK; j = j + 1) begin
            $display("HW OUTPUT %0d = %0d", j + 1, hw_out[j]);
        end

        $display("========================================");
        $display("PERFORMANCE REPORT");
        $display("========================================");
        $display("TOTAL INPUTS             = %0d", input_count);
        $display("TOTAL OUTPUTS            = %0d", output_count);
        $display("TOPK                     = %0d", TOPK);
        $display("START TIME               = %0.0f ns", start_time);
        $display("END TIME                 = %0.0f ns", end_time);
        $display("LATENCY                  = %0.0f ns", latency);
        $display("CLOCK PERIOD             = %0d ns", CLK_PERIOD);
        $display("CLOCK CYCLES             = %0.0f", clock_cycles);
        $display("STREAM INPUT THROUGHPUT  = %0.6f Mbps", stream_throughput_mbps);
        $display("STREAM INPUT THROUGHPUT  = %0.6f Gbps", stream_throughput_gbps);
        $display("OVERALL THROUGHPUT       = %0.6f Mbps", overall_throughput_mbps);
        $display("OVERALL THROUGHPUT       = %0.6f Gbps", overall_throughput_gbps);

        if (error_count == 0) begin
            $display("RESULT                   = PASS");
        end
        else begin
            $display("RESULT                   = FAIL");
            $display("ERROR COUNT              = %0d", error_count);
        end

        $display("========================================");
        $display("SIMULATION FINISHED");
        $display("========================================");

        $finish;
    end
end

endmodule
